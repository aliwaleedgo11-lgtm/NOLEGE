        function switchCompany(company) {
            currentCompany = company;
            currentSubCategory = 'الكل';
            
            document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
            document.getElementById(`nav-${company}`).classList.add('active');

            const nameDisp = document.getElementById('companyNameDisplay');
            const descDisp = document.getElementById('companyDescDisplay');
            const iconDisp = document.getElementById('companyIcon');

            if (company === 'IN-DRIVE') {
                nameDisp.textContent = 'IN DRIVE';
                descDisp.textContent = 'قائمة المشاكل والحلول الخاصة بشركة IN DRIVE';
                iconDisp.className = 'company-icon indrive-bg';
                iconDisp.innerHTML = '<i class="fa-solid fa-car"></i>';
            } else {
                nameDisp.textContent = 'ARWW';
                descDisp.textContent = 'قائمة المشاكل والحلول الخاصة بشركة ARWW';
                iconDisp.className = 'company-icon arww-bg';
                iconDisp.innerHTML = '<i class="fa-solid fa-truck-fast"></i>';
            }

            setupSubCategories();
            renderCards();
        }

        function setupSubCategories() {
            const tabsContainer = document.getElementById('categoryTabs');
            tabsContainer.innerHTML = '';

            const items = database[currentCompany] || [];
            const categories = ['الكل', ...new Set(items.map(i => i.category))];

            categories.forEach(cat => {
                const btn = document.createElement('button');
                btn.className = `sub-tab ${cat === currentSubCategory ? 'active' : ''}`;
                btn.textContent = cat;
                btn.onclick = () => {
                    document.querySelectorAll('.sub-tab').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    currentSubCategory = cat;
                    renderCards(document.getElementById('searchInput').value);
                };
                tabsContainer.appendChild(btn);
            });
        }

        function renderCards(filterText = '') {
            const grid = document.getElementById('cardsGrid');
            grid.innerHTML = '';

            let items = database[currentCompany] || [];

            if (currentSubCategory !== 'الكل') {
                items = items.filter(i => i.category === currentSubCategory);
            }

            const filtered = items.filter(item => 
                item.problem.toLowerCase().includes(filterText.toLowerCase()) ||
                item.solution.toLowerCase().includes(filterText.toLowerCase()) ||
                item.category.toLowerCase().includes(filterText.toLowerCase())
            );

            if (filtered.length === 0) {
                grid.innerHTML = `<div class="empty-state">
                    <i class="fa-solid fa-folder-open" style="font-size: 3rem; margin-bottom: 1rem;"></i>
                    <p>لم يتم العثور على أي نتائج مطابقة للبحث.</p>
                </div>`;
                return;
            }

            filtered.forEach(item => {
                const card = document.createElement('div');
                card.className = 'card';
                card.innerHTML = `
                    <div>
                        <span class="card-category">${item.category}</span>
                        <div class="card-problem">${item.problem}</div>
                        <div class="card-solution">${item.solution}</div>
                    </div>
                    <div class="card-actions">
                        <button class="btn-copy" onclick="copyText(\`${escapeQuotes(item.solution)}\`)">
                            <i class="fa-regular fa-copy"></i> نسخ الحل
                        </button>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function updateBadges() {
            document.getElementById('badge-IN-DRIVE').textContent = database['IN-DRIVE'].length;
            document.getElementById('badge-ARWW').textContent = database['ARWW'].length;
        }

        function handleSearch() {
            const query = document.getElementById('searchInput').value;
            renderCards(query);
        }

        function copyText(text) {
            navigator.clipboard.writeText(text).then(() => {
                const toast = document.getElementById('toast');
                toast.classList.add('show');
                setTimeout(() => {
                    toast.classList.remove('show');
                }, 2500);
            });
        }

        function escapeQuotes(str) {
            return str.replace(/'/g, "\\'").replace(/"/g, '\\"').replace(/\n/g, '\\n');
        }

        function toggleTheme() {
            const body = document.body;
            const icon = document.getElementById('themeIcon');
            
            if (body.getAttribute('data-theme') === 'dark') {
                body.removeAttribute('data-theme');
                icon.className = 'fa-solid fa-moon';
            } else {
                body.setAttribute('data-theme', 'dark');
                icon.className = 'fa-solid fa-sun';
            }
        }
